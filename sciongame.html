// --- ENGINE SETUP --- const { Engine, Render, Runner, World, Bodies, Composite } = Matter; const engine = Engine.create(); const world = engine.world; world.gravity.y = 1.2; // Slightly higher gravity for faster water const render = Render.create({ element: document.getElementById('game-container'), engine: engine, options: { width: window.innerWidth, height: window.innerHeight, wireframes: false, // CRITICAL: Set to false to see images background: '#aed6f1' // Light blue sky background } }); Render.run(render); Runner.run(Runner.create(), engine); // --- LEVEL DATA --- const levels = [ // Level 1 { pipe: { x: 100, y: 100 }, scion: { x: window.innerWidth - 150, y: window.innerHeight - 150 }, obstacles: [{x: window.innerWidth/2, y: window.innerHeight/2 + 50, w: 200, h: 20}] }, // ... define 19 more levels here ... ]; let currentLevel = 0; let scionBody; let pipeBody; // --- LOAD LEVEL --- function loadLevel(levelIdx) { World.clear(world); Engine.clear(engine); // Scion with Image scionBody = Bodies.rectangle(levels[levelIdx].scion.x, levels[levelIdx].scion.y, 100, 100, { isStatic: true, render: { sprite: { texture: 'assets/scion_normal.png', xScale: 0.5, // Adjust based on your image size yScale: 0.5 } } }); // Pipe with Image pipeBody = Bodies.rectangle(levels[levelIdx].pipe.x, levels[levelIdx].pipe.y, 60, 60, { isStatic: true, render: { sprite: { texture: 'assets/pipe.png', xScale: 0.5, yScale: 0.5 } } }); World.add(world, [scionBody, pipeBody]); // Add Obstacles with nice colors levels[levelIdx].obstacles.forEach(obs => { World.add(world, Bodies.rectangle(obs.x, obs.y, obs.w, obs.h, { isStatic: true, render: { fillStyle: '#8e44ad', strokeStyle: 'white', lineWidth: 3 } // Purple obstacles })); }); } // --- ATTRACTIVE DRAWING (Fluid Lines) --- let isDrawing = false; window.addEventListener('mousedown', () => { isDrawing = true; }); window.addEventListener('mouseup', () => { isDrawing = false; }); window.addEventListener('mousemove', (e) => { if (isDrawing) { // Create smaller, more frequent dots for a "line" effect const dot = Bodies.circle(e.clientX, e.clientY, 8, { isStatic: true, render: { fillStyle: '#3498db' } }); World.add(world, dot); } }); // --- FLUID WATER --- function releaseWater() { for (let i = 0; i < 60; i++) { setTimeout(() => { const drop = Bodies.circle(levels[currentLevel].pipe.x, levels[currentLevel].pipe.y, 6, { restitution: 0.3, friction: 0.001, render: { fillStyle: 'rgba(52, 152, 219, 0.8)' } // Semi-transparent blue }); World.add(world, drop); }, i * 30); } } // Spacebar to release water window.addEventListener('keydown', (e) => { if(e.code === "Space") releaseWater(); }); function nextLevel() { currentLevel = (currentLevel + 1) % levels.length; loadLevel(currentLevel); } // Start Game loadLevel(currentLevel);
